cmake_minimum_required(VERSION 3.16)
project(trilayer_nav2_planner)
find_package(ament_cmake REQUIRED)
foreach(dep nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib nav_msgs geometry_msgs tf2_ros)
 find_package(${dep} REQUIRED)
endforeach()
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(${PROJECT_NAME} SHARED src/planner.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib nav_msgs geometry_msgs tf2_ros)
target_link_libraries(${PROJECT_NAME} OpenSSL::Crypto nlohmann_json::nlohmann_json)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 set(test_stage "${CMAKE_CURRENT_BINARY_DIR}/test_install")
 configure_file(test/stage_install.cmake.in stage_install.cmake @ONLY)
 add_test(NAME nav2_private_stage_setup COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_BINARY_DIR}/stage_install.cmake)
 set_tests_properties(nav2_private_stage_setup PROPERTIES FIXTURES_SETUP nav2_private_stage TIMEOUT 30)
 add_executable(plugin_load_contract test/plugin_load.cpp)
 ament_target_dependencies(plugin_load_contract nav2_core pluginlib)
 add_test(NAME nav2_planner_plugin_load_contract COMMAND plugin_load_contract)
 set_tests_properties(nav2_planner_plugin_load_contract PROPERTIES TIMEOUT 10)
 set(TRILAYER_NAV2_TEST_SOLVER_ROOT "" CACHE PATH "Explicit installed source solver Python directory for real Nav2/P6 test")
 if(TRILAYER_NAV2_TEST_SOLVER_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME nav2_planner_real_p6_action_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/run_real_worker_smoke.py ${TRILAYER_NAV2_TEST_SOLVER_ROOT})
  set_tests_properties(nav2_planner_real_p6_action_contract PROPERTIES TIMEOUT 200 RUN_SERIAL TRUE RESOURCE_LOCK "p6_domain185;p6_domain187")
 endif()
 get_property(package_tests DIRECTORY PROPERTY TESTS)
 foreach(test_name IN LISTS package_tests)
  if(NOT test_name STREQUAL "nav2_private_stage_setup")
   set_tests_properties(${test_name} PROPERTIES FIXTURES_REQUIRED nav2_private_stage
    ENVIRONMENT "AMENT_PREFIX_PATH=${test_stage}:$ENV{AMENT_PREFIX_PATH};TRILAYER_TEST_PREFIX=${test_stage}")
  endif()
 endforeach()
endif()
ament_package()
