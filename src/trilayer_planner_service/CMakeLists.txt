cmake_minimum_required(VERSION 3.16)
project(trilayer_planner_service)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/serve DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME service_protocol_contract COMMAND ${CMAKE_COMMAND} -E env PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH} ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_protocol.py)
 set_tests_properties(service_protocol_contract PROPERTIES TIMEOUT 10)
 set(TRILAYER_SERVICE_TEST_SOLVER_ROOT "" CACHE PATH "Explicit source-built solver installation for real socket tests")
 if(TRILAYER_SERVICE_TEST_SOLVER_ROOT)
  add_test(NAME service_socket_session_contract COMMAND ${CMAKE_COMMAND} -E env
   PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH}
   ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_worker.py ${TRILAYER_SERVICE_TEST_SOLVER_ROOT})
  set_tests_properties(service_socket_session_contract PROPERTIES TIMEOUT 80 RUN_SERIAL TRUE RESOURCE_LOCK trilayer_domain186)
 endif()
endif()
ament_package()
